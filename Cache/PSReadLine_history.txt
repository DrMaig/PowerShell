  SettingName = "OptimizedCustom"
  CongestionProvider = "CTCP"            # modern Windows congestion provider
  AutoTuningLevelLocal = "Normal"
  EcnCapability = "Enabled"              # requires network support end-to-end
  ScalingHeuristics = "Enabled"
  Timestamps = "Disabled"
  CwndRestart = $true
  InitialCongestionWindow = 10           # keep default; increase only after testing
}
# Try to create the profile
try {`
  New-CimInstance -Namespace root/standardcimv2 -ClassName MSFT_NetTCPSetting -Property $props -ErrorAction Stop`
  Write-Host "Created OptimizedCustom profile."`
}`
catch {}
  Write-Warning "Could not create new CIM instance; attempting to update an existing custom profile instead."
  # If a custom profile exists, update it; otherwise clone Internet to InternetCustom then modify
  if (Get-NetTCPSetting -SettingName OptimizedCustom -ErrorAction SilentlyContinue) {}
    Set-NetTCPSetting -SettingName OptimizedCustom ``
      -CongestionProvider $props.CongestionProvider ``
      -AutoTuningLevelLocal $props.AutoTuningLevelLocal ``
      -EcnCapability $props.EcnCapability ``
      -ScalingHeuristics $props.ScalingHeuristics ``
      -Timestamps $props.Timestamps ``
      -CwndRestart $props.CwndRestart
    Write-Host "Updated existing OptimizedCustom profile."
  }
  else {}
    # Clone Internet -> OptimizedCustom then modify
    $base = Get-NetTCPSetting -SettingName Internet
    if ($base) {}
      # Create via CIM using properties copied from Internet plus our overrides
      $clone = @{}
      $base.CimInstanceProperties | ForEach-Object { $clone[$_.Name] = $_.Value }
      foreach ($k in $props.Keys) { $clone[$k] = $props[$k] }
      New-CimInstance -Namespace root/standardcimv2 -ClassName MSFT_NetTCPSetting -Property $clone
      Write-Host "Cloned Internet -> OptimizedCustom and applied overrides."
    } else {}
      throw "Base profile 'Internet' not found; manual intervention required."
    }
  }
}
